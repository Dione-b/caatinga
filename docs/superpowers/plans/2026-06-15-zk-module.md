# ZK Module (`@caatinga/zk`) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a Zero-Knowledge module to `caatinga` that lets Stellar developers scaffold, build, prove, and invoke BLS12-381 Groth16 verifiers on Soroban, starting from a multiplier circuit and ending with typed frontend bindings.

**Architecture:** A new `@caatinga/zk` package holds the importable snarkjs→Soroban serialization bridge. The CLI grows thin `zk init`, `zk build`, `zk prove`, and `zk invoke` commands that delegate to that package and to existing `@caatinga/core` contract flows. The verifier contract and circuit boilerplate ship as an editable `zk-starter` template, never as a hidden runtime dependency.

**Tech Stack:** TypeScript/Node (ESM), Circom 2, snarkjs, Rust/Soroban SDK 25.1.0, BLS12-381, Vitest, pnpm workspace.

---

## Decisions locked before implementation

1. **Curve & proof system:** BLS12-381 + Groth16 (matches the official `stellar/soroban-examples/groth16_verifier`).
2. **SDK pin:** `soroban-sdk = "25.1.0"` and `rust-version = "1.89.0"` for the verifier scaffold (the only maintained reference for BLS12-381 on Soroban).
3. **Protocol target:** Protocol 25+. Testnet/mainnet must be confirmed at Protocol 25 before deploy.
4. **Serializer source:** The public Nethermind Privacy Pools repo is BN254-based, so its serializer cannot be vendored. Implement BLS12-381 serialization from scratch using the reference `groth16_verifier/src/test.rs` vectors as the source of truth.
5. **Bootstrap:** `caatinga zk init <projectName>` creates a new project; `caatinga zk init` scaffolds ZK files into an existing project.
6. **Phase 0 caveat:** The verifier scaffold compiles and matches the reference form, but end-to-end proof verification is validated only after Phase 1 serialization + a round-trip test.

```rust
#![no_std]
use soroban_sdk::{
    contract, contracterror, contractimpl, contracttype,
    crypto::bls12_381::{Fr, G1Affine, G2Affine},
    vec, Env, Vec,
};

#[contracterror]
#[derive(Copy, Clone, Debug, Eq, PartialEq, PartialOrd, Ord)]
#[repr(u32)]
pub enum Groth16Error {
    MalformedVerifyingKey = 0,
}

#[derive(Clone)]
#[contracttype]
pub struct VerificationKey {
    pub alpha: G1Affine,
    pub beta: G2Affine,
    pub gamma: G2Affine,
    pub delta: G2Affine,
    pub ic: Vec<G1Affine>,
}

#[derive(Clone)]
#[contracttype]
pub struct Proof {
    pub a: G1Affine,
    pub b: G2Affine,
    pub c: G1Affine,
}

#[contract]
pub struct Groth16Verifier;

#[contractimpl]
impl Groth16Verifier {
    pub fn verify_proof(
        env: Env,
        vk: VerificationKey,
        proof: Proof,
        pub_signals: Vec<Fr>,
    ) -> Result<bool, Groth16Error> {
        let bls = env.crypto().bls12_381();

        if pub_signals.len() + 1 != vk.ic.len() {
            return Err(Groth16Error::MalformedVerifyingKey);
        }

        let mut vk_x = vk.ic.get(0).unwrap();
        for (s, v) in pub_signals.iter().zip(vk.ic.iter().skip(1)) {
            let prod = bls.g1_mul(&v, &s);
            vk_x = bls.g1_add(&vk_x, &prod);
        }

        let neg_a = -proof.a;
        let vp1 = vec![&env, neg_a, vk.alpha, vk_x, proof.c];
        let vp2 = vec![&env, proof.b, vk.beta, vk.gamma, vk.delta];

        Ok(bls.pairing_check(vp1, vp2))
    }
}

mod test;
```

Create `packages/cli/src/commands/zk-init.command.ts`:

```ts
import { type Command } from "commander";
import { createProjectFromTemplate, loadConfig } from "@caatinga/core";
import { runCliAction } from "../utils/errors.js";
import { logger } from "../utils/logger.js";
import { resolveTemplateDir } from "../utils/template-path.js";
import path from "node:path";
import fs from "node:fs/promises";

const ZK_TEMPLATE = "zk-starter";

export function registerZkInitCommand(program: Command): void {
  program
    .command("zk init [projectName]")
    .description("Scaffold a new ZK-enabled project or add ZK files to the current project")
    .option("-t, --template <template>", "Template name", ZK_TEMPLATE)
    .action(async (projectName: string | undefined, options: { template: string }) => {
      await runCliAction(async () => {
        const templateDir = await resolveTemplateDir(options.template);

        if (projectName) {
          const targetDir = path.resolve(projectName);
          await createProjectFromTemplate({
            projectName,
            targetDir,
            templateDir,
          });
          logger.success(`Created ZK project at ${targetDir}`);
          return;
        }

        // In-place scaffold: require an existing caatinga project.
        const cwd = process.cwd();
        let config;
        try {
          config = await loadConfig({ cwd });
        } catch {
          throw new Error(
            "No caatinga.config.ts found in the current directory. Run `caatinga zk init <projectName>` to create a new project."
          );
        }

        // Copy only ZK-specific files into the existing project.
        await createProjectFromTemplate({
          projectName: config.project,
          targetDir: cwd,
          templateDir,
          // Filter to only circuits/ and contracts/verifier/ paths.
          filter: (relativePath: string) =>
            relativePath.startsWith("circuits/") ||
            relativePath.startsWith("contracts/verifier/"),
        });

        logger.success("Added ZK circuit and verifier scaffold to the current project");
      });
    });
}
```

> Note: If `createProjectFromTemplate` does not accept a `filter` option, add that capability in `packages/core/src/templates/create-project-from-template.ts` as part of this task.

```ts
/**
 * Convert a decimal string to a little-endian unsigned 32-byte array.
 */
export function decimalToLe32(decimal: string): Uint8Array {
  let value = BigInt(decimal);
  const out = new Uint8Array(32);
  for (let i = 0; i < 32; i++) {
    out[i] = Number(value & 0xffn);
    value >>= 8n;
  }
  return out;
}

/**
 * Convert a decimal string to a big-endian unsigned 48-byte array.
 */
export function decimalToBe48(decimal: string): Uint8Array {
  let value = BigInt(decimal);
  const out = new Uint8Array(48);
  for (let i = 47; i >= 0; i--) {
    out[i] = Number(value & 0xffn);
    value >>= 8n;
  }
  return out;
}
```

Create `packages/zk/src/serialization/serialize-proof.ts`:

```ts
import { decimalToBe48 } from "./bigint-helpers.js";

export type SerializedG1 = {
  x: Uint8Array; // 48 bytes, big-endian
  y: Uint8Array; // 48 bytes, big-endian
};

export type SerializedG2 = {
  x: [Uint8Array, Uint8Array]; // each 48 bytes, big-endian
  y: [Uint8Array, Uint8Array]; // each 48 bytes, big-endian
};

export type SerializedProof = {
  a: SerializedG1;
  b: SerializedG2;
  c: SerializedG1;
};

export type SnarkjsProof = {
  pi_a: [string, string, string];
  pi_b: [[string, string], [string, string]];
  pi_c: [string, string, string];
  protocol: string;
  curve: string;
};

export function serializeProof(proof: SnarkjsProof): SerializedProof {
  if (proof.curve !== "bls12381") {
    throw new Error(`Expected curve bls12381, got ${proof.curve}`);
  }

  return {
    a: {
      x: decimalToBe48(proof.pi_a[0]),
      y: decimalToBe48(proof.pi_a[1]),
    },
    b: {
      x: [decimalToBe48(proof.pi_b[0][1]), decimalToBe48(proof.pi_b[0][0])],
      y: [decimalToBe48(proof.pi_b[1][1]), decimalToBe48(proof.pi_b[1][0])],
    },
    c: {
      x: decimalToBe48(proof.pi_c[0]),
      y: decimalToBe48(proof.pi_c[1]),
    },
  };
}
```

> Coordinate ordering for G2 follows the reference verifier: `Fq2::new(c0, c1)` where snarkjs emits `pi_b[0] = [c1, c0]` for the x coordinate.

Create `packages/zk/src/serialization/serialize-vk.ts`:

```ts
import { decimalToBe48 } from "./bigint-helpers.js";
import type { SerializedG1, SerializedG2 } from "./serialize-proof.js";

export type SerializedVk = {
  alpha: SerializedG1;
  beta: SerializedG2;
  gamma: SerializedG2;
  delta: SerializedG2;
  ic: SerializedG1[];
};

export type SnarkjsVk = {
  protocol: string;
  curve: string;
  vk_alpha_1: [string, string, string];
  vk_beta_2: [[string, string], [string, string]];
  vk_gamma_2: [[string, string], [string, string]];
  vk_delta_2: [[string, string], [string, string]];
  vk_ic: Array<[string, string, string]>;
};

function g1FromSnarkjs(p: [string, string, string]): SerializedG1 {
  return {
    x: decimalToBe48(p[0]),
    y: decimalToBe48(p[1]),
  };
}

function g2FromSnarkjs(p: [[string, string], [string, string]]): SerializedG2 {
  return {
    x: [decimalToBe48(p[0][1]), decimalToBe48(p[0][0])],
    y: [decimalToBe48(p[1][1]), decimalToBe48(p[1][0])],
  };
}

export function serializeVk(vk: SnarkjsVk): SerializedVk {
  if (vk.curve !== "bls12381") {
    throw new Error(`Expected curve bls12381, got ${vk.curve}`);
  }

  return {
    alpha: g1FromSnarkjs(vk.vk_alpha_1),
    beta: g2FromSnarkjs(vk.vk_beta_2),
    gamma: g2FromSnarkjs(vk.vk_gamma_2),
    delta: g2FromSnarkjs(vk.vk_delta_2),
    ic: vk.vk_ic.map(g1FromSnarkjs),
  };
}
```

```ts
export function ptauSizeForConstraints(constraintCount: number): number {
  // Next power of two above log2(constraintCount) + 1, bounded by available ptau files.
  const needed = Math.ceil(Math.log2(constraintCount)) + 1;
  return Math.min(Math.max(needed, 8), 28);
}
```

```ts
import { runCliAction } from "../utils/errors.js";
import { logger } from "../utils/logger.js";

export function registerZkBuildCommand(program: Command): void {
  program
    .command("zk build [circuitName]")
    .description("Compile a Circom circuit and run the trusted setup")
    .option("--embed-vk", "Emit a static vk.rs artifact for the verifier contract")
    .action(async (circuitName: string | undefined, options: { embedVk: boolean }) => {
      await runCliAction(async () => {
        const config = await loadConfig();
        const zk = config.zk;
        if (!zk || Object.keys(zk.circuits).length === 0) {
          throw new Error("No ZK circuits configured in caatinga.config.ts");
        }

        const name = circuitName ?? Object.keys(zk.circuits)[0];
        const circuit = zk.circuits[name];
        if (!circuit) {
          throw new Error(`Circuit "${name}" not found in caatinga.config.ts`);

        logger.info(`Building circuit "${name}" with curve ${circuit.curve}...`);
        await buildCircuit({
          circuitName: name,
          circuitPath: circuit.path,
          artifactsDir: `.artifacts/zk/${name}`,
          embedVk: options.embedVk,
        });
        logger.success(`Built circuit "${name}"`);
        logger.warn("This is a single-party development trusted setup. Do not use it in production.");
      });
```
